No Terminales Principales (EXP, TERM, POT, VALOR)
No Terminales Recursivos / Prima (EXP_P, TERM_P, POT_P)
Terminales / Operandos (IDV, CNU, CAD, etc.)
Épsilon 'e' (Fin de rama / cadena vacía)
Paréntesis ( Retorno prioritario a EXP ) 
graph TD
    classDef nt fill:#1e3a8a,stroke:#3b82f6,stroke-width:2px,color:#ffffff;
    classDef prima fill:#4338ca,stroke:#6366f1,stroke-width:2px,color:#ffffff;
    classDef terminal fill:#065f46,stroke:#10b981,stroke-width:2px,color:#ffffff;
    classDef eps fill:#374151,stroke:#9ca3af,stroke-width:1px,color:#e5e7eb,stroke-dasharray: 4 4;
    classDef ret fill:#7c2d12,stroke:#f97316,stroke-width:2px,color:#ffffff;

    EXP["EXP (Expresión General)"]:::nt

    %% Nivel EXP
    EXP --> TERM["TERM (Término)"]:::nt
    EXP --> EXP_P["EXP_P (¿Más sumas o restas?)"]:::prima

    %% Ramas de EXP_P
    EXP_P -->|"Opción +"| EP_SUMA["OPA+ (+)  -->  TERM  -->  EXP_P"]:::prima
    EXP_P -->|"Opción -"| EP_RESTA["OPA- (-)  -->  TERM  -->  EXP_P"]:::prima
    EXP_P -->|"Opción e"| EP_EPS["e (Fin de sumas/restas)"]:::eps

    %% Nivel TERM
    TERM --> POT["POT (Potencia)"]:::nt
    TERM --> TERM_P["TERM_P (¿Más mult o div?)"]:::prima

    %% Ramas de TERM_P
    TERM_P -->|"Opción *"| TP_MULT["OPA* (*)  -->  POT  -->  TERM_P"]:::prima
    TERM_P -->|"Opción /"| TP_DIV["OPA/ (/)  -->  POT  -->  TERM_P"]:::prima
    TERM_P -->|"Opción e"| TP_EPS["e (Fin de mult/div)"]:::eps

    %% Nivel POT
    POT --> VALOR["VALOR (Operando Base)"]:::nt
    POT --> POT_P["POT_P (¿Exponente?)"]:::prima

    %% Ramas de POT_P
    POT_P -->|"Opción ^"| PP_POT["OPA^ (^)  -->  POT"]:::prima
    POT_P -->|"Opción e"| PP_EPS["e (Fin de potencia)"]:::eps

    %% Ramas de VALOR
    VALOR --> V_IDV["IDV (Variable)"]:::terminal
    VALOR --> V_CNU["CNU (Número)"]:::terminal
    VALOR --> V_CAD["CAD (Cadena)"]:::terminal
    VALOR --> V_CAR["CAR (Carácter)"]:::terminal
    VALOR --> V_BOOL["PR20-PR22 (Booleanos)"]:::terminal
    VALOR --> V_FUNC["CALL_FUNC (Función)"]:::terminal
    VALOR --> V_PAR["CE1 '('  -->  EXP  -->  CE2 ')'"]:::retorno

    %% Retorno de los paréntesis
    V_PAR -.->|"Evalúa primero la expresión interna"| EXP
